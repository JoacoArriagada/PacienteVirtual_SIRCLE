Consulta
SELECT f.escenario_id, f.id, r.id, v.id, v.version, f.rol, f.ruta, r.sha256, r.contenido, v.estado
FROM versiones_fuentes AS vf
JOIN fuentes AS f ON f.id = vf.fuente_id
JOIN revisiones_fuente AS r ON r.id = vf.revision_fuente_id
JOIN versiones_clinicas AS v ON v.id = vf.version_clinica_id
WHERE v.version LIKE 'originales-%'
ORDER BY f.escenario_id, f.ruta;
